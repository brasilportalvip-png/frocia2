import { useEffect, useRef, useState, useSyncExternalStore } from 'react';
import type { AnchorHTMLAttributes, CSSProperties, ReactNode } from 'react';

const IMAGES: Record<string, string> = __FORJA_IMAGES__;
const RAW_SITE_URL = '__SITE_URL__';

const isBrowser = typeof window !== 'undefined';

/** Endereço público do site (preenchido na publicação). */
export const SITE_URL: string = /^https?:\/\//.test(RAW_SITE_URL)
  ? RAW_SITE_URL.replace(/\/$/, '')
  : isBrowser && /^https?:$/.test(window.location.protocol)
    ? window.location.origin
    : '';

// ------------------------------------------------------------------ fotos

function hashOf(text: string): number {
  let h = 0;
  for (let i = 0; i < text.length; i++) h = (h * 31 + text.charCodeAt(i)) >>> 0;
  return h;
}

/** Fundo elegante enquanto a foto não existe (nunca mostra imagem quebrada). */
export function fallbackPhoto(id: string): string {
  const a = hashOf(id) % 360;
  const b = (a + 38) % 360;
  const svg =
    "<svg xmlns='http://www.w3.org/2000/svg' viewBox='0 0 1600 1000' preserveAspectRatio='xMidYMid slice'>" +
    "<defs><radialGradient id='g' cx='28%' cy='22%' r='95%'>" +
    "<stop offset='0' stop-color='hsl(" + a + ",42%,42%)'/>" +
    "<stop offset='0.55' stop-color='hsl(" + b + ",38%,18%)'/>" +
    "<stop offset='1' stop-color='hsl(" + b + ",30%,8%)'/>" +
    "</radialGradient></defs><rect width='1600' height='1000' fill='url(#g)'/></svg>";
  return 'data:image/svg+xml;charset=utf-8,' + encodeURIComponent(svg);
}

/** URL da foto declarada em forja/site.json pelo id. */
export function photo(id: string): string {
  return IMAGES[id] || fallbackPhoto(id);
}

export function Photo(props: {
  id: string;
  alt: string;
  className?: string;
  style?: CSSProperties;
  priority?: boolean;
}) {
  const { id, alt, className = '', style, priority = false } = props;
  const ref = useRef<HTMLImageElement>(null);
  const [loaded, setLoaded] = useState(false);
  const [failed, setFailed] = useState(false);
  useEffect(() => {
    if (ref.current && ref.current.complete && ref.current.naturalWidth > 0) setLoaded(true);
  }, []);
  return (
    <img
      ref={ref}
      src={failed ? fallbackPhoto(id) : photo(id)}
      alt={alt}
      loading={priority ? 'eager' : 'lazy'}
      decoding="async"
      fetchPriority={priority ? 'high' : 'auto'}
      onLoad={() => setLoaded(true)}
      onError={() => setFailed(true)}
      style={style}
      className={className + ' transition-[opacity,filter] duration-700 ' + (loaded ? 'opacity-100 blur-0' : 'opacity-0 blur-md')}
    />
  );
}

// ------------------------------------------------------------------ rotas

const listeners = new Set<() => void>();

function historyWorks(): boolean {
  if (!isBrowser || !/^https?:$/.test(window.location.protocol)) return false;
  try {
    window.history.replaceState(window.history.state, '');
    return true;
  } catch {
    // Prévia isolada: o histórico do navegador não pode ser usado aqui.
    return false;
  }
}

const useHistory = historyWorks();
let memoryPath = '/';

function cleanPath(path: string): string {
  const p = (path || '/').split('?')[0].split('#')[0] || '/';
  const withSlash = p.startsWith('/') ? p : '/' + p;
  return withSlash.length > 1 ? withSlash.replace(/\/+$/, '') : '/';
}

function readPath(): string {
  return useHistory ? cleanPath(window.location.pathname) : memoryPath;
}

if (useHistory) {
  window.addEventListener('popstate', () => listeners.forEach((l) => l()));
}

function subscribe(listener: () => void) {
  listeners.add(listener);
  return () => {
    listeners.delete(listener);
  };
}

/** Caminho atual ("/", "/sobre", "/privacidade"...). */
export function useRoute(): string {
  return useSyncExternalStore(subscribe, readPath, () => '/');
}

export function navigate(to: string) {
  const hashIndex = to.indexOf('#');
  const anchor = hashIndex >= 0 ? to.slice(hashIndex + 1) : '';
  const pathPart = hashIndex >= 0 ? to.slice(0, hashIndex) : to;
  const before = readPath();
  const target = pathPart ? cleanPath(pathPart) : before;
  if (target !== before) {
    if (useHistory) window.history.pushState(null, '', target + (anchor ? '#' + anchor : ''));
    else memoryPath = target;
    listeners.forEach((l) => l());
  }
  window.setTimeout(() => {
    const el = anchor ? document.getElementById(anchor) : null;
    if (el) el.scrollIntoView({ behavior: 'smooth', block: 'start' });
    else if (target !== before) window.scrollTo({ top: 0, behavior: 'auto' });
  }, 40);
}

type LinkProps = { to: string; children?: ReactNode; onNavigate?: () => void } & Omit<
  AnchorHTMLAttributes<HTMLAnchorElement>,
  'href' | 'onClick'
>;

export function Link(props: LinkProps) {
  const { to, children, onNavigate, ...rest } = props;
  const external = /^(https?:|mailto:|tel:|\/\/)/i.test(to);
  const newTab = /^(https?:|\/\/)/i.test(to);
  return (
    <a
      {...rest}
      href={to}
      target={newTab ? '_blank' : rest.target}
      rel={newTab ? 'noopener noreferrer' : rest.rel}
      onClick={(e) => {
        if (external) {
          if (onNavigate) onNavigate();
          return;
        }
        if (e.metaKey || e.ctrlKey || e.shiftKey || e.altKey || e.button !== 0) return;
        e.preventDefault();
        if (onNavigate) onNavigate();
        navigate(to);
      }}
    >
      {children}
    </a>
  );
}

// ------------------------------------------------------------------ SEO por página

function setMeta(attr: 'name' | 'property', key: string, value: string | undefined) {
  if (!value) return;
  let el = document.head.querySelector('meta[' + attr + '="' + key + '"]') as HTMLMetaElement | null;
  if (!el) {
    el = document.createElement('meta');
    el.setAttribute(attr, key);
    document.head.appendChild(el);
  }
  el.setAttribute('content', value);
}

function setCanonical(href: string) {
  if (!href) return;
  let el = document.head.querySelector('link[rel="canonical"]') as HTMLLinkElement | null;
  if (!el) {
    el = document.createElement('link');
    el.setAttribute('rel', 'canonical');
    document.head.appendChild(el);
  }
  el.setAttribute('href', href);
}

/** Título, descrição, link canônico e cartões de compartilhamento de cada página. */
export function useSeo(meta: { title: string; description?: string; image?: string }) {
  const path = useRoute();
  useEffect(() => {
    document.title = meta.title;
    const url = SITE_URL ? SITE_URL + (path === '/' ? '/' : path) : '';
    const image = meta.image ? (/^(https?:|data:)/.test(meta.image) ? meta.image : photo(meta.image)) : undefined;
    setMeta('name', 'description', meta.description);
    setMeta('property', 'og:title', meta.title);
    setMeta('property', 'og:description', meta.description);
    setMeta('property', 'og:type', 'website');
    setMeta('property', 'og:url', url);
    if (image && !image.startsWith('data:')) {
      setMeta('property', 'og:image', image);
      setMeta('name', 'twitter:image', image);
    }
    setMeta('name', 'twitter:card', 'summary_large_image');
    setMeta('name', 'twitter:title', meta.title);
    setMeta('name', 'twitter:description', meta.description);
    setCanonical(url);
  }, [meta.title, meta.description, meta.image, path]);
}

// ------------------------------------------------------------------ animações e luz

function prefersReducedMotion(): boolean {
  return isBrowser && !!window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
}

const REVEAL_FROM: Record<string, string> = {
  up: 'translate-y-10',
  down: '-translate-y-10',
  left: '-translate-x-10',
  right: 'translate-x-10',
  zoom: 'scale-95',
  fade: '',
};

/** Faz o conteúdo surgir com suavidade quando entra na tela. */
export function Reveal(props: {
  children?: ReactNode;
  delay?: number;
  from?: 'up' | 'down' | 'left' | 'right' | 'zoom' | 'fade';
  className?: string;
  as?: 'div' | 'section' | 'article' | 'li' | 'span' | 'header' | 'footer' | 'figure';
  id?: string;
}) {
  const { children, delay = 0, from = 'up', className = '', as = 'div', id } = props;
  const Tag = as as 'div';
  const ref = useRef<HTMLDivElement>(null);
  const [shown, setShown] = useState(false);
  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    if (prefersReducedMotion() || !('IntersectionObserver' in window)) {
      setShown(true);
      return;
    }
    const io = new IntersectionObserver(
      (entries) => {
        if (entries.some((e) => e.isIntersecting)) {
          setShown(true);
          io.disconnect();
        }
      },
      { threshold: 0, rootMargin: '0px 0px -8% 0px' },
    );
    io.observe(el);
    return () => io.disconnect();
  }, []);
  return (
    <Tag
      ref={ref}
      id={id}
      style={{ transitionDelay: delay + 'ms' }}
      className={
        className +
        ' transition-all duration-[900ms] ease-[cubic-bezier(.2,.7,.2,1)] will-change-transform ' +
        (shown ? 'opacity-100 translate-x-0 translate-y-0 scale-100' : 'opacity-0 ' + REVEAL_FROM[from])
      }
    >
      {children}
    </Tag>
  );
}

/** Luz que acompanha o cursor. Coloque dentro de um bloco "relative overflow-hidden". */
export function Spotlight(props: { color?: string; size?: number; className?: string }) {
  const { color = 'rgba(255,255,255,0.14)', size = 560, className = '' } = props;
  const ref = useRef<HTMLDivElement>(null);
  useEffect(() => {
    const el = ref.current;
    const parent = el ? el.parentElement : null;
    if (!el || !parent || prefersReducedMotion()) return;
    const move = (e: MouseEvent) => {
      const r = parent.getBoundingClientRect();
      el.style.setProperty('--x', e.clientX - r.left + 'px');
      el.style.setProperty('--y', e.clientY - r.top + 'px');
      el.style.opacity = '1';
    };
    const leave = () => {
      el.style.opacity = '0.55';
    };
    parent.addEventListener('mousemove', move);
    parent.addEventListener('mouseleave', leave);
    return () => {
      parent.removeEventListener('mousemove', move);
      parent.removeEventListener('mouseleave', leave);
    };
  }, []);
  return (
    <div
      ref={ref}
      aria-hidden="true"
      className={'pointer-events-none absolute inset-0 transition-opacity duration-500 ' + className}
      style={{ opacity: 0.55, background: 'radial-gradient(' + size + 'px circle at var(--x, 50%) var(--y, 35%), ' + color + ', transparent 62%)' }}
    />
  );
}

/** Movimento sutil de profundidade na rolagem (use em fotos de fundo). */
export function useParallax(strength = 0.12) {
  const ref = useRef<HTMLDivElement>(null);
  useEffect(() => {
    const el = ref.current;
    if (!el || prefersReducedMotion()) return;
    let frame = 0;
    const update = () => {
      frame = 0;
      const r = el.getBoundingClientRect();
      const offset = (r.top + r.height / 2 - window.innerHeight / 2) * -strength;
      el.style.transform = 'translate3d(0,' + offset.toFixed(1) + 'px,0) scale(1.08)';
    };
    const onScroll = () => {
      if (!frame) frame = window.requestAnimationFrame(update);
    };
    update();
    window.addEventListener('scroll', onScroll, { passive: true });
    return () => {
      window.removeEventListener('scroll', onScroll);
      if (frame) window.cancelAnimationFrame(frame);
    };
  }, [strength]);
  return ref;
}

/** true quando a página já rolou (para o menu ganhar fundo de vidro). */
export function useScrolled(threshold = 24): boolean {
  const [scrolled, setScrolled] = useState(false);
  useEffect(() => {
    const on = () => setScrolled(window.scrollY > threshold);
    on();
    window.addEventListener('scroll', on, { passive: true });
    return () => window.removeEventListener('scroll', on);
  }, [threshold]);
  return scrolled;
}

/** Barra fina de progresso de leitura no topo. */
export function ScrollProgress(props: { className?: string }) {
  const ref = useRef<HTMLDivElement>(null);
  useEffect(() => {
    const on = () => {
      const max = document.documentElement.scrollHeight - window.innerHeight;
      if (ref.current) ref.current.style.transform = 'scaleX(' + (max > 0 ? window.scrollY / max : 0) + ')';
    };
    on();
    window.addEventListener('scroll', on, { passive: true });
    return () => window.removeEventListener('scroll', on);
  }, []);
  return (
    <div
      ref={ref}
      aria-hidden="true"
      className={'fixed left-0 top-0 z-[70] h-[3px] w-full origin-left ' + (props.className || 'bg-white')}
      style={{ transform: 'scaleX(0)' }}
    />
  );
}

// ------------------------------------------------------------------ LGPD

export function CookieConsent(props: { brand: string; className?: string }) {
  const [open, setOpen] = useState(false);
  useEffect(() => {
    try {
      setOpen(!window.localStorage.getItem('forja-consent'));
    } catch {
      // Sem armazenamento disponível: mostra o aviso a cada visita.
      setOpen(true);
    }
  }, []);
  const save = (value: string) => {
    try {
      window.localStorage.setItem('forja-consent', value + ':' + Date.now());
    } catch {
      // Sem armazenamento disponível: a escolha vale só nesta visita.
      setOpen(false);
    }
    setOpen(false);
  };
  if (!open) return null;
  return (
    <div
      role="dialog"
      aria-label="Aviso de cookies"
      className={
        'fixed inset-x-4 bottom-4 z-[80] mx-auto max-w-3xl rounded-2xl border border-white/10 bg-neutral-950/90 p-5 text-sm text-neutral-200 shadow-2xl backdrop-blur-xl md:flex md:items-center md:gap-6 ' +
        (props.className || '')
      }
    >
      <p className="flex-1 leading-relaxed">
        {props.brand} usa cookies essenciais para o site funcionar e, com a sua permissão, cookies de medição para melhorar a sua experiência. Veja a{' '}
        <Link to="/privacidade" className="font-medium text-white underline underline-offset-4">
          Política de Privacidade
        </Link>
        .
      </p>
      <div className="mt-4 flex shrink-0 gap-2 md:mt-0">
        <button type="button" onClick={() => save('essenciais')} className="rounded-full border border-white/20 px-4 py-2 transition hover:bg-white/10">
          Só essenciais
        </button>
        <button type="button" onClick={() => save('todos')} className="rounded-full bg-white px-4 py-2 font-semibold text-neutral-900 transition hover:bg-neutral-200">
          Aceitar
        </button>
      </div>
    </div>
  );
}

type LegalSection = { title: string; body: ReactNode[] };

function LegalDocument(props: { sections: LegalSection[]; updated: string; className?: string }) {
  return (
    <div className={'mx-auto max-w-3xl space-y-10 text-base leading-relaxed ' + (props.className || '')}>
      <p className="text-sm opacity-70">Última atualização: {props.updated}</p>
      {props.sections.map((s, i) => (
        <section key={s.title}>
          <h2 className="mb-3 text-xl font-semibold tracking-tight">
            {i + 1}. {s.title}
          </h2>
          <div className="space-y-3 opacity-90">
            {s.body.map((b, j) => (
              <p key={j}>{b}</p>
            ))}
          </div>
        </section>
      ))}
    </div>
  );
}

function updatedLabel(): string {
  try {
    return new Date(document.lastModified || Date.now()).toLocaleDateString('pt-BR', { day: '2-digit', month: 'long', year: 'numeric' });
  } catch {
    // Navegador sem formatação de datas em português.
    return new Date().getFullYear().toString();
  }
}
